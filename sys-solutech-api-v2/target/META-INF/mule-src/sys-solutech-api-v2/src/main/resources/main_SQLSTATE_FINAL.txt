<?xml version="1.0" encoding="UTF-8"?>

<mule xmlns:ee="http://www.mulesoft.org/schema/mule/ee/core"
      xmlns:http="http://www.mulesoft.org/schema/mule/http"
      xmlns="http://www.mulesoft.org/schema/mule/core"
      xmlns:doc="http://www.mulesoft.org/schema/mule/documentation"
      xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
      xsi:schemaLocation="
          http://www.mulesoft.org/schema/mule/ee/core http://www.mulesoft.org/schema/mule/ee/core/current/mule-ee.xsd
          http://www.mulesoft.org/schema/mule/http http://www.mulesoft.org/schema/mule/http/current/mule-http.xsd
          http://www.mulesoft.org/schema/mule/core http://www.mulesoft.org/schema/mule/core/current/mule.xsd">

    <!-- 🚀 Flow to post customers to Solutech ERP -->
    <flow name="post:\v1\integrations\customers:application\json:solutech-system-main-config" doc:id="ef988ccc-9674-4862-91e4-e159436c229f">

        <!-- 1️⃣ Listener -->

        <!-- 2️⃣ Capture incoming payload -->
        <set-variable variableName="receivedPayload" value="#[payload]" doc:name="receivedPayload"/>
        <logger level="INFO" doc:name="Log Input"
                message="📦 Payload being sent to Solutech: #[write(payload, 'application/json', {indent:2})]" />

        <!-- 3️⃣ Get OAuth Token -->
        <flow-ref 
            doc:name="Get Token"
            name="post:\oauth\token:application\x-www-form-urlencoded:solutech-system-main-config" />

        <!-- 4️⃣ Try posting to Solutech -->
<try doc:name="Try - POST to Solutech ERP">
    <http:request method="POST"
                  config-ref="HTTP_Request_Solutech_Token_config"
                  path="/api/v1/integrations/customers"
                  doc:name="POST to Solutech ERP">
        <http:body><![CDATA[#[vars.receivedPayload]]]></http:body>
        <http:headers><![CDATA[#[{
            'Authorization': 'Bearer ' ++ vars.access_token,
            'Content-Type': 'application/json'
        }]]]></http:headers>
        <http:response-validator>
            <http:success-status-code-validator values="0..599"/>
        </http:response-validator>
    </http:request>

    <!-- Pass through Solutech's raw response and status -->
    <set-variable value="#[payload]" doc:name="solutechPayload" doc:id="9f84fe75-467b-4696-ac1f-fc6752ef738d" variableName="solutechPayload"/>
			<set-variable variableName="solutechStatus" value="#[attributes.statusCode]" doc:name="statusCode"/>
    <set-variable variableName="solutechReason" value="#[attributes.reasonPhrase]" doc:name="solutechReason"/>

    <logger level="INFO" doc:name="Log Response" doc:id="95b1c30e-b3d1-43a8-a499-d3c7dc5bd317" message="📦 Payload being received from Solutech: #[vars.solutechPayload]"/>
<choice doc:name="Choice Router - Return Solutech Response">

    <!-- 🟢 Success -->
<when expression="#[(vars.solutechStatus default 0) &gt;= 200 and (vars.solutechStatus default 0) &lt; 300]">
        <logger level="INFO" message="🟢 Solutech Success → Code: #[vars.solutechStatus], Response: #[vars.solutechPayload]" />
        <ee:transform doc:name="Return Success JSON">
            <ee:message>
                <ee:set-payload><![CDATA[
%dw 2.0
output application/json
---
readUrl("classpath://json/successResponse.json", "application/json")
                ]]></ee:set-payload>
                <ee:set-attributes><![CDATA[
%dw 2.0
output application/java
---
{
    statusCode: 200,
    reasonPhrase: "OK",
    headers: {
        "Content-Type": "application/json"
    }
}
                ]]></ee:set-attributes>
            </ee:message>
        </ee:transform>
    </when>

    <!-- 🔴 400 Error -->
    <when expression="#[(vars.solutechStatus default 0) == 400]">
        <logger level="INFO" message="🔴 Solutech Error 400 → #[vars.solutechPayload]" />
        <ee:transform doc:name="Return 400 JSON">
            <ee:message>
                <ee:set-payload><![CDATA[
%dw 2.0
output application/json
---
readUrl("classpath://json/error400Response.json", "application/json")
                ]]></ee:set-payload>
                <ee:set-attributes><![CDATA[
%dw 2.0
output application/java
---
{
    statusCode: 400,
    reasonPhrase: "Bad Request",
    headers: {
        "Content-Type": "application/json"
    }
}
                ]]></ee:set-attributes>
            </ee:message>
        </ee:transform>
    </when>

    <!-- 🟠 401 Error -->
    <when expression="#[(vars.solutechStatus default 0) == 401]">
        <logger level="INFO" message="🟠 Solutech Error 401 → #[vars.solutechPayload]" />
        <ee:transform doc:name="Return 401 JSON">
            <ee:message>
                <ee:set-payload><![CDATA[
%dw 2.0
output application/json
---
readUrl("classpath://json/error401Response.json", "application/json")
                ]]></ee:set-payload>
                <ee:set-attributes><![CDATA[
%dw 2.0
output application/java
---
{
    statusCode: 401,
    reasonPhrase: "Unauthorized",
    headers: {
        "Content-Type": "application/json"
    }
}
                ]]></ee:set-attributes>
            </ee:message>
        </ee:transform>
    </when>

    <!-- 🔵 404 Error -->
    <when expression="#[(vars.solutechStatus default 0) == 404]">
        <logger level="INFO" message="🔵 Solutech Error 404 → #[vars.solutechPayload]" />
        <ee:transform doc:name="Return 404 JSON">
            <ee:message>
                <ee:set-payload><![CDATA[
%dw 2.0
output application/json
---
readUrl("classpath://json/error404Response.json", "application/json")
                ]]></ee:set-payload>
                <ee:set-attributes><![CDATA[
%dw 2.0
output application/java
---
{
    statusCode: 404,
    reasonPhrase: "Not Found",
    headers: {
        "Content-Type": "application/json"
    }
}
                ]]></ee:set-attributes>
            </ee:message>
        </ee:transform>
    </when>

    <!-- ⚫ 500 Error -->
    <otherwise>
        <logger level="INFO" message="⚫ Solutech Error 500 or Unknown → #[vars.solutechPayload]" />
        <ee:transform doc:name="Return 500 JSON">
            <ee:message>
                <ee:set-payload><![CDATA[
%dw 2.0
output application/json
---
readUrl("classpath://json/error500Response.json", "application/json")
                ]]></ee:set-payload>
                <ee:set-attributes><![CDATA[
%dw 2.0
output application/java
---
{
    statusCode: 500,
    reasonPhrase: "Internal Server Error",
    headers: {
        "Content-Type": "application/json"
    }
}
                ]]></ee:set-attributes>
            </ee:message>
        </ee:transform>
    </otherwise>

</choice>
			<error-handler >
				<on-error-continue type="ANY">
    <logger level="ERROR" message="Uncaught Error: #[error.description]" />
    <set-payload value="#[{message: 'Unexpected error', detail: error.description}]" />
</on-error-continue>
			</error-handler>

</try>
   

        <!-- 5️⃣ Final Log -->
        <logger 
            level="INFO"
            doc:name="Log Response"
            message="🟢 Solutech Response → Status: #[attributes.statusCode], Body: #[vars.solutechPayload]" />

    </flow>

    <!-- 🔑 Token Retrieval Subflow -->
    <sub-flow name="post:\oauth\token:application\x-www-form-urlencoded:solutech-system-main-config"
              doc:id="487e502c-4a8e-4063-ab56-a00081d5e666">

        <logger level="INFO"
                message="Client Secret: #[if (p('env') == 'dev') p('secure::erp.solutechlabs.client_secret') else '********']" />

        <until-successful maxRetries="${http.request.retry.count}"
                          millisBetweenRetries="${http.request.retry.interval}"
                          doc:name="Until Successful">

            <http:request method="POST"
                          path="/oauth/token"
                          config-ref="HTTP_Request_Solutech_Token_config"
                          doc:name="Request">

                <http:body><![CDATA[#[%dw 2.0
output application/x-www-form-urlencoded
---
{
    grant_type: p('erp.solutechlabs.grant_type'),
    client_id: p('erp.solutechlabs.client_id'),
    client_secret: p('secure::erp.solutechlabs.client_secret'),
    scope: p('erp.solutechlabs.scope')
}]]]></http:body>

                <http:headers><![CDATA[#[{
                    'Content-Type': 'application/x-www-form-urlencoded'
                }]]]></http:headers>

            </http:request>
        </until-successful>

        <set-variable variableName="access_token" value="#[payload.access_token]" />
        <logger level="INFO" doc:name="access_token" message="access_token: #[vars.access_token]" />
    </sub-flow>

</mule>
