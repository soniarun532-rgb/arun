<flow name="post:\v1\integrations\customers:application\json:solutech-system-main-config"
      doc:id="84c1c562-3aad-426a-85d6-ccdb8746eb0c">

    <!-- 🟢 Log start -->
    <logger level="INFO"
            message='#["Starting the " ++ flow.name ++ " at " ++ now() as String]'
            doc:name="Start Logger"/>

    <!-- 🔑 Get OAuth Token -->
    <flow-ref name="post:\oauth\token:application\x-www-form-urlencoded:solutech-system-main-config"
              doc:name="Get Access Token"/>

    <!-- 🔁 Attempt ERP request -->
    <until-successful maxRetries="${http.request.retry.count}"
                      millisBetweenRetries="${http.request.retry.interval}"
                      doc:name="Until Successful">

        <try doc:name="ERP Request Try">

            <!-- 🌍 HTTP POST to Solutech ERP -->
            <http:request method="#[vars.request_method]"
                          config-ref="HTTP_Request_Solutech_Token_config"
                          path="#['/api/v1/integrations/' ++ vars.request_endpoint]"
                          sendBodyMode="ALWAYS"
                          doc:name="Post Customer">
                <http:body><![CDATA[#[vars.customerPayload]]]></http:body>
                <http:headers><![CDATA[#[{
                    'Authorization': 'Bearer ' ++ vars.access_token,
                    'Content-Type': 'application/json'
                }]]]></http:headers>
            </http:request>

            <!-- 🧾 Capture ERP response -->
            <set-variable variableName="httpStatus"
                          value="#[attributes.statusCode]"
                          doc:name="Capture HTTP Status"/>

            <choice doc:name="Handle ERP Response">
                <when expression="#[vars.httpStatus == 400]">
                    <logger level="WARN"
                            message="⚠️ ERP returned 400 Bad Request → #[payload]"
                            doc:name="Log 400 Response"/>
                    <set-variable variableName="httpReason" value="Bad Request"/>
                    <set-variable variableName="responseHeaders" value='{"Content-Type": "text/plain"}'/>
                </when>
                <otherwise>
                    <logger level="INFO"
                            message="✅ ERP responded #[vars.httpStatus] → #[payload]"
                            doc:name="Log Success Response"/>
                    <set-variable variableName="httpReason" value="OK"/>
                    <set-variable variableName="responseHeaders" value='{"Content-Type": "application/json"}'/>
                </otherwise>
            </choice>

            <!-- ⚠️ Handle HTTP 400 locally -->
            <error-handler>
                <on-error-continue type="HTTP:BAD_REQUEST"
                                   logException="false"
                                   doc:name="Handle 400 Bad Request">
                    <logger level="WARN"
                            message="🚫 Caught 400 Bad Request → returning raw ERP error body"
                            doc:name="Log 400 Catch"/>
                    <set-variable variableName="httpStatus" value="400"/>
                    <set-variable variableName="httpReason" value="Bad Request"/>
                    <set-variable variableName="responseHeaders" value='{"Content-Type": "text/plain"}'/>
                    <set-payload value="#[(error.detailedDescription default error.description default error.message)]"
                                 doc:name="Set ERP Error Body"/>
                </on-error-continue>
            </error-handler>

        </try>
    </until-successful>

    <!-- 📦 Return final response -->
    <logger level="INFO"
            message="✅ Finished ERP call for #[vars.request_endpoint] → Status #[vars.httpStatus]"
            doc:name="End Logger"/>

</flow>
