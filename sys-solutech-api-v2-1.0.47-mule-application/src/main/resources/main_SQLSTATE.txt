<?xml version="1.0" encoding="UTF-8"?>

<mule xmlns:ee="http://www.mulesoft.org/schema/mule/ee/core"
      xmlns:http="http://www.mulesoft.org/schema/mule/http"
      xmlns="http://www.mulesoft.org/schema/mule/core"
      xmlns:doc="http://www.mulesoft.org/schema/mule/documentation"
      xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
      xsi:schemaLocation="
          http://www.mulesoft.org/schema/mule/ee/core http://www.mulesoft.org/schema/mule/ee/core/current/mule-ee.xsd
          http://www.mulesoft.org/schema/mule/http http://www.mulesoft.org/schema/mule/http/current/mule-http.xsd
          http://www.mulesoft.org/schema/mule/core http://www.mulesoft.org/schema/mule/core/current/mule.xsd">

    <!-- 🚀 Flow to post customers to Solutech ERP -->
    <flow name="solutech_post_flow" doc:id="a81a7b36-694c-4266-9c33-4366336aaf9f">

        <!-- 1️⃣ Listener -->
        <http:listener 
            doc:name="Listener"
            config-ref="solutech-system-api-httpListenerConfig"
            path="/integrations/customers" />

        <!-- 2️⃣ Capture incoming payload -->
        <set-variable variableName="receivedPayload" value="#[payload]" />
        <logger level="INFO" doc:name="Log Input"
                message="📦 Payload being sent to Solutech: #[write(payload, 'application/json', {indent:2})]" />

        <!-- 3️⃣ Get OAuth Token -->
        <flow-ref 
            doc:name="Get Token"
            name="post:\oauth\token:application\x-www-form-urlencoded:solutech-system-main-config" />

        <!-- 4️⃣ Try posting to Solutech -->
<try doc:name="Try - POST to Solutech ERP">
    <http:request method="POST"
                  config-ref="HTTP_Request_Solutech_Token_config"
                  path="/api/v1/integrations/customers"
                  doc:name="POST to Solutech ERP">
        <http:body><![CDATA[#[vars.receivedPayload]]]></http:body>
        <http:headers><![CDATA[#[{
            'Authorization': 'Bearer ' ++ vars.access_token,
            'Content-Type': 'application/json'
        }]]]></http:headers>
        <http:response-validator>
            <http:success-status-code-validator values="0..599"/>
        </http:response-validator>
    </http:request>

    <!-- Pass through Solutech's raw response and status -->
    <set-variable value="#[payload]" doc:name="solutechPaylod" doc:id="961422fa-2288-45cc-bf13-9a84898558a1" variableName="solutechPaylod"/>
			<set-variable variableName="solutechStatus" value="#[attributes.statusCode]" doc:name="statusCode"/>
    <set-variable variableName="solutechReason" value="#[attributes.reasonPhrase]" doc:name="solutechReason"/>

    <choice doc:name="Choice" doc:id="4bb69f0f-e8b5-40e8-afeb-0444eda26a7d" >
			<when expression="#[(vars.solutechPayload startsWith '{') or (vars.solutechPayload startsWith '[')]">
					<ee:transform doc:name="Return Same Response to Postman">
        <ee:message>
            <ee:set-payload><![CDATA[vars.solutechPaylod]]></ee:set-payload>

            <ee:set-attributes><![CDATA[
                %dw 2.0
                output application/java
                ---
                {
                    statusCode: vars.solutechStatus,
                    reasonPhrase: vars.solutechReason,
                    headers: {
                        "Content-Type": "text/plain"
                    }
                }
            ]]></ee:set-attributes>
        </ee:message>
    </ee:transform>
				</when>
				<otherwise >
					<ee:transform doc:name="Transform Message" doc:id="3cc8686a-fe4f-4a30-852f-1423d3d5fd20" >
						<ee:message >
							<ee:set-payload ><![CDATA[vars.solutechPaylod]]></ee:set-payload>
						</ee:message>
					</ee:transform>
				</otherwise>
			</choice>
</try>
   

        <!-- 5️⃣ Final Log -->
        <logger 
            level="INFO"
            doc:name="Log Response"
            message="🟢 Solutech Response → Status: #[attributes.statusCode], Body: #[vars.solutechPaylod]" />

    </flow>

    <!-- 🔑 Token Retrieval Subflow -->
    <sub-flow name="post:\oauth\token:application\x-www-form-urlencoded:solutech-system-main-config"
              doc:id="2757b7d3-98fb-4308-a1f6-78b0b7d3f286">

        <logger level="INFO"
                message="Client Secret: #[if (p('env') == 'dev') p('secure::erp.solutechlabs.client_secret') else '********']" />

        <until-successful maxRetries="${http.request.retry.count}"
                          millisBetweenRetries="${http.request.retry.interval}"
                          doc:name="Until Successful">

            <http:request method="POST"
                          path="/oauth/token"
                          config-ref="HTTP_Request_Solutech_Token_config"
                          doc:name="Request">

                <http:body><![CDATA[#[%dw 2.0
output application/x-www-form-urlencoded
---
{
    grant_type: p('erp.solutechlabs.grant_type'),
    client_id: p('erp.solutechlabs.client_id'),
    client_secret: p('secure::erp.solutechlabs.client_secret'),
    scope: p('erp.solutechlabs.scope')
}]]]></http:body>

                <http:headers><![CDATA[#[{
                    'Content-Type': 'application/x-www-form-urlencoded'
                }]]]></http:headers>

            </http:request>
        </until-successful>

        <set-variable variableName="access_token" value="#[payload.access_token]" />
        <logger level="INFO" doc:name="access_token" message="access_token: #[vars.access_token]" />
    </sub-flow>

    <!-- ⚠️ ERP Error Handler -->
    <sub-flow name="handle-erp-error">
        <logger level="WARN" message="⚠️ Handling ERP Error #[error.errorType.identifier]: #[error.description]" />

        <set-variable variableName="erpErrorBody" value="#[error.errorResponse.body default error.description]" />
        <set-variable variableName="statusCode"
                      value="#[
                        if (error.errorType.identifier startsWith 'HTTP:BAD_REQUEST') 400
                        else if (error.errorType.identifier startsWith 'HTTP:UNAUTHORIZED') 401
                        else if (error.errorType.identifier startsWith 'HTTP:FORBIDDEN') 403
                        else if (error.errorType.identifier startsWith 'HTTP:NOT_FOUND') 404
                        else 500
                      ]"/>

        <set-payload value='#[%dw 2.0
            output application/json
            ---
            {
                statusCode: vars.statusCode,
                statusMessage: error.description default "ERP Error",
                details: vars.erpErrorBody as String,
                timestamp: now() as String
            }]'/>

        <logger level="WARN" message="🚫 ERP Error Response JSON: #[payload]" />
    </sub-flow>

</mule>
